\begin{recipeDP}
    [
        preparationtime = {\SI{20}{\minute}},
        bakingtime = {\SI{75}{\minute}},
        bakingtemperature = {\protect\bakingtemperature{fanoven=\SI{180}{\celsius}}},
        portion = {23er Springform},
        source = {thebananadiaries.com}
    ]
    {Blaubeer-Zitronen-Kuchen}

    \graph
        {
            small=Gebaeck/Kuchen/Blaubeer-Zitronen-Kuchen/small.jpg,
            big=Gebaeck/Kuchen/Blaubeer-Zitronen-Kuchen/big.jpg
        }

    % \introduction{
    %     Ein weicher, saftiger Zitronen-Kaffeekuchen mit frischen Heidelbeeren, cremiger Frischkäse-Füllung und leichter Zimtkrümel-Oberschicht.
    % }

    \ingredients{
        \multicolumn{2}{l}{\textbf{Krümel}} \\
        \SI{125}{\g} & Mehl \\
        \SI{1}{\TL} & Zimt \\
        \SI{150}{\g} & Zucker \\
        \SI{70}{\g} & Butter \\
        \SI{1}{\pck} & Vanillezucker \\
        1 Prise & Meersalz \\
        \\
        \multicolumn{2}{l}{\textbf{Frischkäse-Füllung}} \\
        \SI{300}{\g} & Frischkäse \\
        \SI{10}{\g} & Zucker \\
        \SI{1}{\EL} & Zitronenzeste\index{Zitrone!Zeste} \\
        \SI{1}{\EL} & Speisestärke \\
        \\
        \multicolumn{2}{l}{\textbf{Teig}} \\
        \SI{300}{\g} & \addtoidx{Heidelbeeren} \\
        \SI{375}{\g} & Mehl \\
        \SI{1,5}{\TL} & Backpulver \\
        \SI[parse-numbers = false]{\nicefrac{1}{2}}{\TL} & Natron \\
        \SI[parse-numbers = false]{\nicefrac{1}{4}}{\TL} & Salz \\
        \SI{250}{\g} & Zucker \\
        \SI{113}{\g} & Butter \\
        \SI{170}{\g} & Joghurt \\
        \SI{80}{\ml} & Milch \\
        \SI{60}{\ml} & Zitronensaft \\
        \SI{3}{\EL} & Zitronenzeste \\
        \SI{1}{\EL} & Vanilleextrakt \\
        \\
        \multicolumn{2}{l}{\textbf{Glasur}} \\
        \SI{140}{\g} & Puderzucker \\
        \SI{2}{\EL} & Milch \\
        \SI{1}{\pck} & Vanillezucker
    }

    \preparation{
        \step Den Ofen auf \SI{180}{\celsius} vorheizen und eine Springform (Durchmesser \SI{23}{\cm}) mit Backpapier auslegen. Die Ränder leicht einfetten.

        \step Heidelbeeren kurz abspülen und mit \SI{1}{\EL} Mehl in einer kleinen Schüssel vermengen.
        Für die Krümel alle Zutaten mit einer Gabel verrühren, bis eine sandige Konsistenz entsteht.

        \step Die Frischkäse-Füllung mit Zucker, Zitronenzeste und Speisestärke verrühren, bis alles cremig ist.

        \step Für den Teig Zucker und Zitronenzeste mit den Fingerspitzen sanft reiben.
        Dann Butter, Joghurt und Vanilleextrakt unterrühren.

        \step Mehl, Backpulver, Natron und Salz vermischen und zur Zuckermischung geben.
        Nach und nach Milch und Zitronensaft unterrühren, dann die Heidelbeeren vorsichtig unterheben.

        \step Die Hälfte des Teigs in die Form geben, die Frischkäse-Füllung darauf verteilen und mit der restlichen Teigmasse bedecken.
        Die Krümel darüber streuen.

        \step Den Kuchen etwa \SI{75}{\minute} backen, bis ein Zahnstocher sauber herauskommt.
        Anschließend abkühlen lassen.

        \step Für die Glasur Puderzucker, Milch und Vanillezucker verrühren und über den abgekühlten Kuchen gießen.
        Vor dem Servieren kurz ruhen lassen.
    }

\end{recipeDP}
